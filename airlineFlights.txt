#include <iostream>
#include <fstream>
#include <sstream>
#include <string>
#include <map>
#include <vector>
using namespace std;

int main()
{
    ifstream fin("airports.txt");
    map<string, vector<string>> airportMap;
    string line;
    getline(fin, line); // Skip column headers
    while (getline(fin, line))
    {
        string origin, destination;
        stringstream ss(line);
        getline(ss, origin, ',');
        getline(ss, destination, ',');
        airportMap[origin].push_back(destination);
    }

fin.close();
return 0;
}
